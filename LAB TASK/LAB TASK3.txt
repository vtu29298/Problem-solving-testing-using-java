import java.util.Scanner;

public class MaximumProfitAnalyzer {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter number of days: ");
        int n = sc.nextInt();

        int[] price = new int[n];

        System.out.println("Enter stock prices:");
        for (int i = 0; i < n; i++) {
            price[i] = sc.nextInt();
        }

        int minPrice = price[0];
        int maxProfit = 0;
        int buyDay = 0;
        int sellDay = 0;

        for (int i = 1; i < n; i++) {
            int profit = price[i] - minPrice;

            if (profit > maxProfit) {
                maxProfit = profit;
                sellDay = i;
            }

            if (price[i] < minPrice) {
                minPrice = price[i];
                buyDay = i;
            }
        }

        System.out.println("\n--- Maximum Profit Analysis ---");
        System.out.println("Buy Day   : " + (buyDay + 1));
        System.out.println("Sell Day  : " + (sellDay + 1));
        System.out.println("Buy Price : " + price[buyDay]);
        System.out.println("Sell Price: " + price[sellDay]);
        System.out.println("Maximum Profit: " + maxProfit);

        sc.close();
    }
}
